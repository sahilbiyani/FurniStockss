One-time Supabase setup

Before publishing this version, enable Phone provider in Supabase Authentication and run the SQL below in SQL Editor. Replace the placeholder phone with the one authorised editor number in E.164 format. Do not enter it in the website code.

Show secure database SQL 
-- Safe to run on an existing FurniStock database.
-- Adds the new columns without deleting existing stock records.
alter table public.furniture_stock
  add column if not exists unit text not null default '',
  add column if not exists remark text not null default '',
  add column if not exists returned_quantity integer not null default 0;

-- Keep returned quantity within the amount already consumed.
do $$
begin
  if not exists (
    select 1 from pg_constraint
    where conrelid = 'public.furniture_stock'::regclass
      and conname = 'furniture_stock_returned_quantity_check'
  ) then
    alter table public.furniture_stock
      add constraint furniture_stock_returned_quantity_check
      check (returned_quantity >= 0 and returned_quantity <= consumed_quantity);
  end if;
end $$;

-- Remove any old policies from the stock table.
do $$
declare p record;
begin
  for p in select policyname from pg_policies
           where schemaname='public' and tablename='furniture_stock'
  loop
    execute format('drop policy if exists %I on public.furniture_stock', p.policyname);
  end loop;
end $$;

create table if not exists public.furniture_editors (
  phone text primary key check (phone ~ '^\+[1-9][0-9]{7,14}$')
);
alter table public.furniture_editors enable row level security;

do $$
declare p record;
begin
  for p in select policyname from pg_policies
           where schemaname='public' and tablename='furniture_editors'
  loop
    execute format('drop policy if exists %I on public.furniture_editors', p.policyname);
  end loop;
end $$;

revoke all on public.furniture_editors from anon, authenticated;
grant select on public.furniture_editors to authenticated;

create policy "editor can read own phone row"
on public.furniture_editors for select to authenticated
using (phone = (auth.jwt() ->> 'phone'));

alter table public.furniture_stock enable row level security;
revoke all on public.furniture_stock from anon;
grant select, insert, update, delete on public.furniture_stock to authenticated;
grant usage, select on sequence public.furniture_stock_id_seq to authenticated;

create policy "signed in users can view stock"
on public.furniture_stock for select to authenticated
using (auth.uid() is not null);

create policy "allowlisted phone can insert stock"
on public.furniture_stock for insert to authenticated
with check (exists (
  select 1 from public.furniture_editors e
  where e.phone = (auth.jwt() ->> 'phone')
));

create policy "allowlisted phone can update stock"
on public.furniture_stock for update to authenticated
using (exists (
  select 1 from public.furniture_editors e
  where e.phone = (auth.jwt() ->> 'phone')
))
with check (exists (
  select 1 from public.furniture_editors e
  where e.phone = (auth.jwt() ->> 'phone')
));

create policy "allowlisted phone can delete stock"
on public.furniture_stock for delete to authenticated
using (exists (
  select 1 from public.furniture_editors e
  where e.phone = (auth.jwt() ->> 'phone')
));

-- The authorised editor phone, in E.164 format (India +91).
insert into public.furniture_editors(phone)
values ('+919011078809')
on conflict (phone) do nothing;

-- Add to Realtime only if it is not already published.
do $$
begin
  if not exists (
    select 1 from pg_publication_tables
    where pubname = 'supabase_realtime'
      and schemaname = 'public'
      and tablename = 'furniture_stock'
  ) then
    alter publication supabase_realtime add table public.furniture_stock;
  end if;
end $$;

If the SQL reports that the realtime table is already in the publication, that part is already configured. Keep the editor allowlist table private; never grant public/anon access to it.
